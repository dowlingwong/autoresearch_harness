%======================== Row 1 ========================
\begin{panel}[315mm]
  {1. Task scores evaluate results,\\not experimental conduct}
  {What must hold $\rightarrow$ what we measure}
  \color{black}\raggedright
  A higher AUC never shows whether the agent stayed in scope or who decided
  acceptance. Each condition becomes a reported metric.

  \entry{D1 \, Bounded edit scope $\rightarrow$ invalid-action rate}{No
  protected benchmark, data or evaluation file changes.}
  \entry{D2 \, Complete attempt accounting $\rightarrow$ failure taxonomy}{Every
  attempt recorded --- invalid proposals and crashes.}
  \entry{D3 \, Independent adjudication $\rightarrow$ acceptance rate}{Fixed
  control-plane rules decide, never the agent.}
  \entry{D4 \, Provenance linkage $\rightarrow$ provenance completeness}{Each
  attempt links to patch, config, run, metric, decision.}
  \entry{D5 \, Recomputability $\rightarrow$ artifact manifest}{Claims
  recompute from the ledger; no training rerun.}

  \par\medskip
  {\bfseries Governance metrics are primary; task AUC is secondary.\par}
\end{panel}
\begin{panel}[315mm]
  {2. The governed control plane}
  {Agent $=$ Model $+$ Harness}
  \color{black}\raggedright

  \par\smallskip\centerline{\archdiagram}\par\medskip
  \cpt{\textbf{Memory is an optional review channel; the ledger is the
  authoritative audit record} --- owned by the control plane. Managers and
  workers are swappable; the contract is fixed.}
\end{panel}
\begin{panel}[315mm]
  {3. The governed trial contract}
  {Exactly one immutable record per trial}
  \color{black}\raggedright

  \par\smallskip\centerline{\lifecyclediagram}\par\medskip

  \callout{Every opened budget slot appends \textbf{exactly one} terminal
  record before the next may open.}

  \cpt{\textbf{Structural, not empirical:} the invariant holds by
  construction. 100\% provenance over 1{,}445 trials is its certificate.\\[1.5mm]
  \textbf{Not claimed:} any AUC gain --- it sits within seed noise.\\[1.5mm]
  \textbf{Label check:} $\kappa = 0.830$ on 50 failed-invalid trials.}
\end{panel}

\begin{panel}[315mm]
  {5. Next: predicting failure}
  {R1 --- failure-dynamics benchmark}
  \color{black}\raggedright

  \par\smallskip\centerline{\roadmapstrip}\par\medskip

  \cpt{The audit record shows \emph{that} a trial failed --- not \emph{when}
  the trajectory started to go wrong.}

  \entry{Step-granular contract}{Typed per-step events: state differences,
  evidence provenance, memory reads.}
  \entry{Causal fault injection}{Seven mechanism families, deterministic
  injectors, a clean counterfactual per fault.}
  \entry{Calibrated early warning}{$q_t = P(\text{terminal failure} \mid
  H_t)$ and a cause distribution. Reported: hazard, propagation, lead time,
  recoverability window.}

  \par\medskip
  {\bfseries No replicated failure law and no warning gain $\Rightarrow$
  released as benchmark plus negative result.\par}
\end{panel}
\begin{panel}[315mm]
  {6. Next: acting on the warning}
  {R2 --- cost-sensitive verified recovery}
  \color{black}\raggedright

  \cpt{A warning still does not say whether evidence is worth buying, or
  which recovery is safe.}

  \entry{Belief in, action out}{From frozen R1: failure hazard, cause,
  evidence freshness, reversibility, damage.}
  \entry{Action space}{continue $\cdot$ verify $\cdot$ recover (retry /
  repair / rollback / re-plan) $\cdot$ escalate $\cdot$ abstain}
  \entry{Matched-budget comparison}{Against never, always, periodic, random
  and confidence-threshold verification.}
  \entry{Process supervision}{Train only on independently adjudicated
  recovery transitions.}

  \callout{Same rule throughout: nothing enters the record unless an
  authority other than the agent verified it.}
\end{panel}
